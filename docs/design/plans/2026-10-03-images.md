# Image Optimisation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `?image` / `?jsx` imports that produce optimised responsive variants at build time, a `<Picture>` component that emits minimal modern `<picture>`/`<img>` markup, CDN URL builders, and a raw descriptor API.

**Architecture:** Pure planner + markup builder in `@rivetjs/runtime` (`image-plan.ts`, `image-markup.ts`, `image-cdn.ts`), wrapped by server (`image.ts`) and DOM (`image-dom.ts`) entries. Build side in `@rivetjs/compiler` (`images/*.ts`): directive parser, sharp processor, disk cache, pipeline, and a second Vite plugin returned by `rivet({ images })` that shares one asset registry across environments and emits every variant from the client environment (app builder orders `ssr` → `client`).

**Tech Stack:** TypeScript (erasable syntax, run by Node 24 type stripping), Vite 8.3.2 environment/builder API, sharp 0.35.5, node:test, Playwright Chromium.

**Spec:** [docs/design/specs/2026-10-03-images-design.md](../specs/2026-10-03-images-design.md)

```ts
// libs/runtime/image-plan.ts
/** Image layout: drives which widths are generated and the default `sizes`. */
export type Layout = "fixed" | "constrained" | "full-width";
export const LAYOUTS: readonly Layout[] = ["fixed", "constrained", "full-width"];
export type ImageFormat = "avif" | "webp" | "jpeg" | "png" | "gif" | "svg";
/** Formats the build can encode to. */
export type EncodableFormat = "avif" | "webp" | "jpeg" | "png";
export const ENCODABLE: readonly EncodableFormat[] = ["avif", "webp", "jpeg", "png"];
/** One `<source>`: `type` absent = format negotiated by the server (CDN auto format). */
export interface ImageSource {
  type?: string;
  srcset: string;
}
export type Placeholder = { kind: "blur"; src: string } | { kind: "color"; value: string };
/** Plain JSON result of `?image` / `cdn()`; the raw API. Sources ordered best → fallback. */
export interface ImageDescriptor {
  src: string;
  width: number;
  height: number;
  layout: Layout;
  sizes?: string;
  placeholder?: Placeholder;
  sources: ImageSource[];
}
export const DEFAULT_BREAKPOINTS: readonly number[] = [640, 768, 1024, 1280, 1536, 1920, 2560];
export const DEFAULT_DENSITIES: readonly number[] = [1, 2];
const MIME: Record<ImageFormat, string> = {
  avif: "image/avif",
  webp: "image/webp",
  jpeg: "image/jpeg",
  png: "image/png",
  gif: "image/gif",
  svg: "image/svg+xml",
};
export function mimeType(format: ImageFormat): string {
  return MIME[format];
}
export interface PlanInput {
  intrinsic: { width: number; height: number };
  layout: Layout;
  width?: number;
  height?: number;
  densities?: readonly number[];
  breakpoints?: readonly number[];
}
export interface PlannedWidth {
  width: number;
  /** srcset descriptor: `640w` or `2x`. */
  descriptor: string;
}
export interface Plan {
  /** Display size (CLS). */
  width: number;
  height: number;
  widths: PlannedWidth[];
  /** Requested width exceeded the intrinsic width. */
  clamped: boolean;
}
/** Widths to generate for a layout; never upscales past the intrinsic width. */
export function planWidths(input: PlanInput): Plan {
  const { intrinsic, layout } = input;
  const densities = [...(input.densities ?? DEFAULT_DENSITIES)].sort((a, b) => a - b);
  const breakpoints = input.breakpoints ?? DEFAULT_BREAKPOINTS;
  const requested = input.width ?? intrinsic.width;
  const cap = intrinsic.width;
  const width = Math.min(requested, cap);
  const aspect = input.height ? input.height / requested : intrinsic.height / intrinsic.width;
  const height = Math.round(width * aspect);
  const clamped = requested > cap;
  if (layout === "fixed") {
    const seen = new Map<number, string>();
    for (const d of densities) {
      const w = Math.min(Math.round(width * d), cap);
      if (!seen.has(w)) seen.set(w, `${+(w / width).toFixed(2)}x`);
    }
    const widths = [...seen].map(([w, descriptor]) => ({ width: w, descriptor })).sort((a, b) => a.width - b.width);
    return { width, height, widths, clamped };
  }
  const max = layout === "full-width" ? cap : Math.min(Math.round(width * densities.at(-1)!), cap);
  const set = new Set(breakpoints.filter((b) => b <= max));
  if (layout === "constrained") for (const d of densities) set.add(Math.min(Math.round(width * d), cap));
  set.add(max);
  const widths = [...set].sort((a, b) => a - b).map((w) => ({ width: w, descriptor: `${w}w` }));
  return { width, height, widths, clamped };
}
/** Default `sizes` for a layout; fixed images use x descriptors and need none. */
export function layoutSizes(layout: Layout, width: number): string | undefined {
  if (layout === "fixed") return undefined;
  return layout === "full-width" ? "100vw" : `(min-width: ${width}px) ${width}px, 100vw`;
}
/** Universally supported last-resort format: PNG keeps alpha, otherwise JPEG (PNG sources stay PNG). */
export function fallbackFormat(source: string, hasAlpha: boolean): EncodableFormat {
  return hasAlpha || source === "png" ? "png" : "jpeg";
}
export interface DescribeInput {
  plan: Plan;
  layout: Layout;
  placeholder?: Placeholder;
  /** Ordered best → fallback; `urls` align with `plan.widths`. */
  sources: { format?: ImageFormat; urls: string[] }[];
}
export function describe({ plan, layout, placeholder, sources }: DescribeInput): ImageDescriptor {
  let index = 0;
  plan.widths.forEach((w, k) => {
    if (w.width <= plan.width) index = k;
  });
  const sizes = layoutSizes(layout, plan.width);
  return {
    src: sources.at(-1)!.urls[index]!,
    width: plan.width,
    height: plan.height,
    layout,
    ...(sizes ? { sizes } : {}),
    ...(placeholder ? { placeholder } : {}),
    sources: sources.map((s) => ({
      ...(s.format ? { type: mimeType(s.format) } : {}),
      srcset: s.urls.map((url, k) => `${url} ${plan.widths[k]!.descriptor}`).join(", "),
    })),
  };
}
/** Descriptor for a file served as-is (SVG, animated GIF/WebP). */
export function single(url: string, format: ImageFormat, width: number, height: number, layout: Layout): ImageDescriptor {
  return { src: url, width, height, layout, sources: [{ type: mimeType(format), srcset: url }] };
}
```

```ts
// libs/runtime/image-markup.ts
import { encodeAttributeValue } from "./template.ts";
import type { ImageDescriptor } from "./image-plan.ts";

/** Tailwind default screens; `sizes` map keys. */
export const SCREENS: Readonly<Record<string, number>> = { sm: 640, md: 768, lg: 1024, xl: 1280, "2xl": 1536 };
/** Raw string, `"auto"`, or map: screen name / px number → `(min-width)`, `(raw media)`, `default`. */
export type SizesProp = string | Record<string, string>;
/** Media query → descriptor; `default` renders the `<img>`. */
export type ArtDirection = { default: ImageDescriptor; [media: string]: ImageDescriptor };
export interface PictureProps {
  image: ImageDescriptor | ArtDirection;
  /** Required; `""` marks a decorative image. */
  alt: string;
  sizes?: SizesProp;
  /** Above-the-fold: eager + `fetchpriority="high"`. Pair with `<ImagePreload>`. */
  priority?: boolean;
  loading?: "lazy" | "eager";
  decoding?: "async" | "sync" | "auto";
  fetchpriority?: "high" | "low" | "auto";
  /** Force `<picture>` or bare `<img>`; default collapses when `<picture>` adds nothing. */
  as?: "picture" | "img";
  class?: string;
  style?: string;
  id?: string;
  title?: string;
  crossorigin?: "anonymous" | "use-credentials";
  referrerpolicy?: string;
  pictureClass?: string;
}
const attr = (name: string, value: string | number | undefined) => encodeAttributeValue(name, value);
const isDescriptor = (value: unknown): value is ImageDescriptor =>
  typeof (value as ImageDescriptor).src === "string" && Array.isArray((value as ImageDescriptor).sources);
const usesWidths = (set: string) => /\s\d+w(?:,|$)/.test(set);
export function resolveSizes(sizes: SizesProp | undefined, fallback: string | undefined): string | undefined {
  if (sizes === undefined) return fallback;
  if (typeof sizes === "string") return sizes === "auto" ? `auto, ${fallback ?? "100vw"}` : sizes;
  const raw: string[] = [];
  const widths: [number, string][] = [];
  let last = fallback;
  for (const [key, value] of Object.entries(sizes)) {
    if (key === "default") last = value;
    else if (key.startsWith("(")) raw.push(`${key} ${value}`);
    else {
      const px = SCREENS[key] ?? (/^\d+$/.test(key) ? Number(key) : NaN);
      if (Number.isNaN(px)) throw new TypeError(`Unknown sizes key "${key}"`);
      widths.push([px, value]);
    }
  }
  widths.sort((a, b) => b[0] - a[0]);
  return [...raw, ...widths.map(([px, value]) => `(min-width: ${px}px) ${value}`), last ?? "100vw"].join(", ");
}
const BLUR = /^data:image\/(?:webp|png|jpeg);base64,[A-Za-z0-9+/]+=*$/;
const COLOR = /^#[0-9a-fA-F]{3,8}$/;
function placeholderStyle(placeholder: ImageDescriptor["placeholder"]): string {
  if (!placeholder) return "";
  if (placeholder.kind === "blur" && BLUR.test(placeholder.src))
    return `background-image:url(${placeholder.src});background-size:cover;background-position:center;background-repeat:no-repeat`;
  if (placeholder.kind === "color" && COLOR.test(placeholder.value)) return `background-color:${placeholder.value}`;
  throw new TypeError("Invalid image placeholder");
}
/** Escaped `<picture>`/`<img>` markup; shared by the server and DOM entries. */
export function pictureHtml(props: PictureProps): string {
  if (typeof props.alt !== "string") throw new TypeError('Picture requires alt (use "" for decorative images)');
  const entries = isDescriptor(props.image)
    ? [{ media: undefined as string | undefined, image: props.image }]
    : [
        ...Object.entries(props.image).filter(([m]) => m !== "default").map(([media, image]) => ({ media: media as string | undefined, image })),
        { media: undefined, image: props.image.default },
      ];
  const main = entries.at(-1)!.image;
  const auto = props.sizes === "auto";
  if (auto && (props.priority || props.loading === "eager")) throw new TypeError('sizes="auto" requires lazy loading');
  if (props.as === "img" && entries.length > 1) throw new TypeError("Art direction requires <picture>");
  const wrap = props.as === "picture" || (props.as !== "img" && (entries.length > 1 || main.sources.length > 1));
  let html = "";
  if (wrap) {
    html += `<picture${attr("class", props.pictureClass)}>`;
    for (const { media, image } of entries) {
      const sizes = resolveSizes(props.sizes, image.sizes);
      for (const source of media ? image.sources : image.sources.slice(0, -1))
        html +=
          `<source${attr("media", media)}${attr("type", source.type)}${attr("srcset", source.srcset)}` +
          `${usesWidths(source.srcset) ? attr("sizes", sizes) : ""}` +
          `${media ? attr("width", image.width) + attr("height", image.height) : ""}>`;
    }
  }
  const fallback = main.sources.at(-1)!;
  const sizes = resolveSizes(props.sizes, main.sizes);
  const style = [placeholderStyle(main.placeholder), props.style].filter(Boolean).join(";");
  html +=
    `<img${attr("src", main.src)}${fallback.srcset === main.src ? "" : attr("srcset", fallback.srcset)}` +
    `${usesWidths(fallback.srcset) ? attr("sizes", sizes) : ""}${attr("width", main.width)}${attr("height", main.height)}` +
    `${attr("alt", props.alt)}${attr("loading", props.priority ? "eager" : auto ? "lazy" : (props.loading ?? "lazy"))}` +
    `${attr("decoding", props.decoding ?? "async")}${attr("fetchpriority", props.priority ? "high" : props.fetchpriority)}` +
    `${attr("class", props.class)}${attr("id", props.id)}${attr("title", props.title)}${attr("crossorigin", props.crossorigin)}` +
    `${attr("referrerpolicy", props.referrerpolicy)}${attr("style", style || undefined)}>`;
  return wrap ? html + "</picture>" : html;
}
export interface PreloadLink {
  rel: "preload";
  as: "image";
  href: string;
  imagesrcset?: string;
  imagesizes?: string;
  type?: string;
  fetchpriority: "high" | "low" | "auto";
  media?: string;
}
/** Preload attributes for the best source (`type` lets unsupported browsers skip it). */
export function preloadLink(
  image: ImageDescriptor,
  options: { sizes?: SizesProp; media?: string; fetchpriority?: "high" | "low" | "auto" } = {},
): PreloadLink {
  const best = image.sources[0]!;
  const sizes = usesWidths(best.srcset) ? resolveSizes(options.sizes, image.sizes) : undefined;
  return {
    rel: "preload",
    as: "image",
    href: image.src,
    ...(best.srcset !== image.src ? { imagesrcset: best.srcset } : {}),
    ...(sizes ? { imagesizes: sizes } : {}),
    ...(best.type ? { type: best.type } : {}),
    fetchpriority: options.fetchpriority ?? "high",
    ...(options.media ? { media: options.media } : {}),
  };
}
export function preloadHtml(link: PreloadLink): string {
  return (
    `<link${attr("rel", link.rel)}${attr("as", link.as)}${attr("href", link.href)}${attr("imagesrcset", link.imagesrcset)}` +
    `${attr("imagesizes", link.imagesizes)}${attr("type", link.type)}${attr("fetchpriority", link.fetchpriority)}${attr("media", link.media)}>`
  );
}
/** srcset of the best source, or of the source with this MIME type. */
export function srcset(image: ImageDescriptor, type?: string): string {
  const source = type === undefined ? image.sources[0] : image.sources.find((s) => s.type === type);
  if (!source) throw new TypeError(`No ${type} source`);
  return source.srcset;
}
const cssUrl = (url: string) => `url("${url.replace(/["\\\n]/g, (c) => (c === "\n" ? "\\a " : "\\" + c))}")`;
/** CSS `image-set()` with 1x/2x candidates per type (for backgrounds). */
export function imageSet(image: ImageDescriptor): string {
  const parts: string[] = [];
  for (const source of image.sources) {
    const entries = source.srcset.split(/,\s+/).map((part) => {
      const [url, d = "1x"] = part.trim().split(/\s+/) as [string, string?];
      return { url, value: Number(d.slice(0, -1)), unit: d.at(-1) };
    });
    const type = source.type ? ` type("${source.type}")` : "";
    if (entries[0]!.unit === "x") {
      for (const e of entries) parts.push(`${cssUrl(e.url)}${type} ${e.value}x`);
      continue;
    }
    const seen = new Set<string>();
    for (const density of [1, 2]) {
      const need = image.width * density;
      const pick = entries.find((e) => e.value >= need) ?? entries.at(-1)!;
      if (seen.has(pick.url)) continue;
      seen.add(pick.url);
      parts.push(`${cssUrl(pick.url)}${type} ${density}x`);
    }
  }
  return `image-set(${parts.join(", ")})`;
}
```

```ts
// libs/runtime/image-cdn.ts
import { planWidths, describe, DEFAULT_DENSITIES, type ImageDescriptor, type ImageFormat, type Layout } from "./image-plan.ts";

export interface CdnUrlOptions {
  width: number;
  height?: number;
  format?: ImageFormat;
  quality?: number;
  fit?: "cover" | "contain";
}
/** URL builder for an image CDN; `autoFormat` = CDN picks the format from `Accept`. */
export interface ImageCdn {
  url(src: string, options: CdnUrlOptions): string;
  autoFormat?: boolean;
}
export interface CdnImageOptions {
  cdn: ImageCdn;
  /** Display size; required so `<img>` reserves space (CLS). */
  width: number;
  height: number;
  layout?: Layout;
  /** Explicit formats; default: none when `autoFormat`, else avif/webp/jpeg. */
  formats?: ImageFormat[];
  quality?: number;
  fit?: "cover" | "contain";
  /** Largest variant; default `width × max(densities)`. */
  maxWidth?: number;
  densities?: number[];
  breakpoints?: number[];
}
/** Descriptor for a CDN-hosted image: pure string building at render time, no fetch. */
export function cdn(src: string, options: CdnImageOptions): ImageDescriptor {
  if (!(options.width > 0) || !(options.height > 0)) throw new TypeError(`cdn("${src}") needs width and height`);
  const densities = options.densities ?? DEFAULT_DENSITIES;
  const maxWidth = options.maxWidth ?? Math.round(options.width * Math.max(...densities));
  const layout = options.layout ?? "constrained";
  const plan = planWidths({
    intrinsic: { width: maxWidth, height: Math.round((maxWidth * options.height) / options.width) },
    layout,
    width: options.width,
    height: options.height,
    densities,
    breakpoints: options.breakpoints,
  });
  const formats: (ImageFormat | undefined)[] = options.formats ?? (options.cdn.autoFormat ? [undefined] : ["avif", "webp", "jpeg"]);
  const fit = options.fit ?? "cover";
  return describe({
    plan,
    layout,
    sources: formats.map((format) => ({
      format,
      urls: plan.widths.map(({ width }) =>
        options.cdn.url(src, { width, height: Math.round((width * plan.height) / plan.width), format, quality: options.quality, fit }),
      ),
    })),
  });
}
const path = (src: string) => src.replace(/^\/+/, "");
const short = (format: ImageFormat | undefined) => (format === "jpeg" ? "jpg" : format);
/** `{src} {width} {height} {format} {quality} {fit}` placeholders; missing values become empty. */
export function urlTemplate(template: string, { autoFormat = false }: { autoFormat?: boolean } = {}): ImageCdn {
  return {
    autoFormat,
    url: (src, o) =>
      template.replace(/\{(src|width|height|format|quality|fit)\}/g, (_, key: keyof CdnUrlOptions | "src") =>
        key === "src" ? path(src) : o[key] == null ? "" : encodeURIComponent(String(o[key])),
      ),
  };
}
export function cloudinary({ cloud }: { cloud: string }): ImageCdn {
  return {
    autoFormat: true,
    url: (src, o) =>
      `https://res.cloudinary.com/${cloud}/image/upload/${[
        `w_${o.width}`,
        o.height && `h_${o.height}`,
        o.height && `c_${o.fit === "contain" ? "fit" : "fill"}`,
        `f_${short(o.format) ?? "auto"}`,
        `q_${o.quality ?? "auto"}`,
      ]
        .filter(Boolean)
        .join(",")}/${path(src)}`,
  };
}
export function imgix({ domain }: { domain: string }): ImageCdn {
  return {
    autoFormat: true,
    url: (src, o) => {
      const q = new URLSearchParams({ w: String(o.width) });
      if (o.height) q.set("h", String(o.height)), q.set("fit", o.fit === "contain" ? "clip" : "crop");
      if (o.format) q.set("fm", short(o.format)!);
      else q.set("auto", "format");
      if (o.quality) q.set("q", String(o.quality));
      return `https://${domain}/${path(src)}?${q}`;
    },
  };
}
export function cloudflare({ origin = "" }: { origin?: string } = {}): ImageCdn {
  return {
    autoFormat: true,
    url: (src, o) =>
      `${origin}/cdn-cgi/image/${[
        `width=${o.width}`,
        o.height && `height=${o.height}`,
        `fit=${o.fit === "contain" ? "contain" : "cover"}`,
        `format=${o.format ?? "auto"}`,
        o.quality && `quality=${o.quality}`,
      ]
        .filter(Boolean)
        .join(",")}/${path(src)}`,
  };
}
```

```ts
import type { ImageProcessor } from "./processor.ts";

export const IMAGE_FILE = /\.(?:jpe?g|png|webp|avif|gif|svg|tiff?)$/i;
export type PlaceholderKind = "blur" | "dominant-color" | "none";
export type Fit = "cover" | "contain" | "inside";
/** Per-import / preset options (`?width=320&layout=fixed&image`). */
export interface Directives {
  layout?: Layout;
  width?: number;
  height?: number;
  formats?: EncodableFormat[];
  quality?: number;
  densities?: number[];
  placeholder?: PlaceholderKind;
  fit?: Fit;
  position?: string;

  layout?: Layout;
  formats?: EncodableFormat[];
  densities?: number[];
  breakpoints?: number[];
  quality?: Partial<Record<EncodableFormat, number>>;
  placeholder?: PlaceholderKind;
  presets?: Record<string, Directives>;
  processor?: ImageProcessor;
  /** Root-relative; default `node_modules/.rivet/images`. */
  cacheDir?: string;
}
export interface ImageOptions {
  layout: Layout;
  width?: number;
  height?: number;
  formats: EncodableFormat[];
  quality: Record<EncodableFormat, number>;
  densities: number[];
  breakpoints: number[];
  placeholder: PlaceholderKind;
  fit: Fit;
  position: string;
}
export const DEFAULT_QUALITY: Record<EncodableFormat, number> = { avif: 50, webp: 75, jpeg: 78, png: 80 };
const PLACEHOLDERS: readonly PlaceholderKind[] = ["blur", "dominant-color", "none"];
const FITS: readonly Fit[] = ["cover", "contain", "inside"];
const POSITIONS = ["center", "top", "right", "bottom", "left", "entropy", "attention"];
const VITE_KEYS = new Set(["import", "t", "v"]);
export interface ImageRequest {
  file: string;
  kind: "image" | "jsx";
  directives: Directives & { preset?: string };
}
/** `null` when `id` is not an image import with the `image`/`jsx` keyword. */
export function parseImageId(id: string): ImageRequest | null {
  const q = id.indexOf("?");
  if (q < 0) return null;
  const file = id.slice(0, q);
  if (!IMAGE_FILE.test(file)) return null;
  const params = new URLSearchParams(id.slice(q + 1));
  const image = params.has("image"),
    jsx = params.has("jsx");
  if (!image && !jsx) return null;
  if (image && jsx) throw new Error(`${file}: use either ?image or ?jsx, not both`);
  const raw: Record<string, string> = {};
  for (const [key, value] of params) if (key !== "image" && key !== "jsx" && !VITE_KEYS.has(key)) raw[key] = value;
  return { file, kind: image ? "image" : "jsx", directives: parseDirectives(raw, file) };
}
const list = (value: string) => value.split(";").filter(Boolean);
export function parseDirectives(raw: Record<string, string>, where: string): Directives & { preset?: string } {
  const out: Directives & { preset?: string } = {};
  for (const [key, value] of Object.entries(raw)) {
    if (key === "width" || key === "height" || key === "quality") out[key] = /^\d+$/.test(value) ? Number(value) : NaN;
    else if (key === "densities") out.densities = list(value).map(Number);
    else if (key === "formats") out.formats = list(value) as EncodableFormat[];
    else if (key === "layout") out.layout = value as Layout;
    else if (key === "placeholder") out.placeholder = value as PlaceholderKind;
    else if (key === "fit") out.fit = value as Fit;
    else if (key === "position" || key === "preset") out[key] = value;
    else throw new Error(`${where}: unknown image directive "${key}"`);
  }
  checkDirectives(out, where);
  return out;
}
const oneOf = (where: string, name: string, value: unknown, allowed: readonly string[]) => {
  if (value !== undefined && !allowed.includes(value as string)) throw new Error(`${where}: ${name} must be one of ${allowed.join(", ")}`);
};
const positiveInt = (value: unknown) => Number.isInteger(value) && (value as number) > 0;
function checkFormats(formats: readonly string[] | undefined, where: string) {
  for (const f of formats ?? []) if (!ENCODABLE.includes(f as EncodableFormat)) throw new Error(`${where}: unknown image format "${f}"`);
}
/** Validate typed directives (from a query string, a preset, or config). */
export function checkDirectives(d: Directives, where: string): Directives {
  for (const key of ["width", "height"] as const)
    if (d[key] !== undefined && !positiveInt(d[key])) throw new Error(`${where}: ${key} must be a positive integer`);
  if (d.quality !== undefined && !(positiveInt(d.quality) && d.quality <= 100)) throw new Error(`${where}: quality must be 1-100`);
  if (d.densities !== undefined && !(d.densities.length && d.densities.every((n) => n > 0 && Number.isFinite(n))))
    throw new Error(`${where}: densities must be positive numbers`);
  checkFormats(d.formats, where);
  oneOf(where, "layout", d.layout, LAYOUTS);
  oneOf(where, "placeholder", d.placeholder, PLACEHOLDERS);
  oneOf(where, "fit", d.fit, FITS);
  oneOf(where, "position", d.position, POSITIONS);
  return d;
}
export function validateConfig(config: ImagesConfig): void {
  const where = "images config";
  checkDirectives({ layout: config.layout, formats: config.formats, densities: config.densities, placeholder: config.placeholder }, where);
  if (config.breakpoints && !config.breakpoints.every(positiveInt)) throw new Error(`${where}: breakpoints must be positive integers`);
  for (const [format, q] of Object.entries(config.quality ?? {})) {
    checkFormats([format], where);
    if (!(positiveInt(q) && q <= 100)) throw new Error(`${where}: quality must be 1-100`);
  }
  for (const [name, preset] of Object.entries(config.presets ?? {})) checkDirectives(preset, `images preset "${name}"`);
}
/** Merge: import directives > preset > config > defaults. */
export function resolveOptions(config: ImagesConfig, directives: Directives & { preset?: string }, file: string): ImageOptions {
  const { preset: name, ...own } = directives;
  const preset = name === undefined ? {} : config.presets?.[name];
  if (!preset) throw new Error(`${file}: unknown image preset "${name}"`);
  const pick = <K extends keyof Directives>(key: K): Directives[K] => own[key] ?? preset[key];
  const quality = pick("quality");
  return {
    layout: pick("layout") ?? config.layout ?? "constrained",
    width: pick("width"),
    height: pick("height"),
    formats: pick("formats") ?? config.formats ?? ["avif", "webp"],
    quality: quality === undefined ? { ...DEFAULT_QUALITY, ...config.quality } : { avif: quality, webp: quality, jpeg: quality, png: quality },
    densities: pick("densities") ?? config.densities ?? [...DEFAULT_DENSITIES],
    breakpoints: config.breakpoints ?? [...DEFAULT_BREAKPOINTS],
    placeholder: pick("placeholder") ?? config.placeholder ?? "blur",
    fit: pick("fit") ?? "cover",
    position: pick("position") ?? "center",
  };
}
```

```ts
export function sharpProcessor(): ImageProcessor {
  let loaded: Promise<Sharp> | undefined;
  const load = () =>
    (loaded ??= import("sharp").then(
      (m) => m.default as unknown as Sharp,
      () => {
        throw new Error("`images` needs `sharp`: npm i -D sharp");
      },
    ));

    version = (createRequire(import.meta.url)("sharp/package.json") as { version: string }).version;
  } catch {}

    id: `sharp@${version}`,
    async metadata(input) {
      const m = await (await load())(input).metadata();
      const format = m.format === "heif" ? (m.compression === "av1" ? "avif" : null) : m.format;
      if (!format || !["jpeg", "png", "webp", "avif", "gif", "svg", "tiff"].includes(format)) throw new Error(`unsupported image format "${m.format}"`);
      const rotated = (m.orientation ?? 1) >= 5;
      return {
        width: (rotated ? m.height : m.width)!,
        height: (rotated ? m.width : m.height)!,
        format: format as ImageMetadata["format"],
        hasAlpha: Boolean(m.hasAlpha),
        animated: (m.pages ?? 1) > 1,
      };
    },
    async encode(input, o) {
      const sharp = await load();
      const p = sharp(input, { animated: false }).rotate().resize({ width: o.width, height: o.height, fit: o.fit, position: o.position, withoutEnlargement: true });
      const out =
        o.format === "avif" ? p.avif({ quality: o.quality, effort: 4 })
        : o.format === "webp" ? p.webp({ quality: o.quality })
        : o.format === "jpeg" ? p.jpeg({ quality: o.quality, mozjpeg: true })
        : p.png({ quality: o.quality, palette: o.quality < 100, compressionLevel: 9 });
      return out.toBuffer();
    },
    async placeholder(input, kind) {
      const sharp = await load();
      if (kind === "dominant-color") {
        const { r, g, b } = (await sharp(input).stats()).dominant;
        return { kind: "color", value: "#" + [r, g, b].map((n) => n.toString(16).padStart(2, "0")).join("") };
      }
      const small = await sharp(input).rotate().resize(16).webp({ quality: 40 }).toBuffer();
      return { kind: "blur", src: `data:image/webp;base64,${small.toString("base64")}` };
    },
```

```ts
// libs/compiler/images/cache.ts
import { mkdir, readFile, rename, writeFile } from "node:fs/promises";
import { createHash, randomUUID } from "node:crypto";
import { availableParallelism } from "node:os";
import { join } from "node:path";

export function hash(...parts: (string | Buffer)[]): string {
  const h = createHash("sha256");
  for (const part of parts) h.update(part);
  return h.digest("hex");
}
/** Content-addressed files; writes are atomic (temp + rename) so parallel builds never read partial output. */
export function diskCache(dir: string) {
  return {
    async get(name: string): Promise<Buffer | undefined> {
      try {
        return await readFile(join(dir, name));
      } catch (error) {
        if ((error as NodeJS.ErrnoException).code === "ENOENT") return undefined;
        throw error;
      }
    },
    async set(name: string, data: Buffer): Promise<void> {
      await mkdir(dir, { recursive: true });
      const temp = join(dir, `${name}.${randomUUID()}.tmp`);
      await writeFile(temp, data);
      await rename(temp, join(dir, name));
    },
  };
}
/** Run at most `limit` tasks at once. */
export function pool(limit = availableParallelism()) {
  let active = 0;
  const queue: (() => void)[] = [];
  return async <T>(task: () => Promise<T>): Promise<T> => {
    if (active >= limit) await new Promise<void>((resolve) => queue.push(resolve));
    active++;
    try {
      return await task();
    } finally {
      active--;
      queue.shift()?.();
    }
  };
}
```

```ts
// libs/compiler/images/pipeline.ts
import { readFile } from "node:fs/promises";
import { basename, extname } from "node:path";
import { describe, fallbackFormat, planWidths, single, type ImageDescriptor, type ImageFormat, type Placeholder } from "@rivetjs/runtime/image-plan";
import { resolveOptions, type Directives, type ImagesConfig } from "./options.ts";
import type { EncodeOptions, ImageProcessor } from "./processor.ts";
import { diskCache, hash, pool } from "./cache.ts";

export interface PlannedAsset {
  key: string;
  ext: string;
  file: string;
  fileName: string;
  url: string;
  /** Absent: copy the source bytes (SVG, animated). */
  encode?: EncodeOptions;
}
export interface Described {
  descriptor: ImageDescriptor;
  assets: PlannedAsset[];
  warnings: string[];
}
const EXT: Record<ImageFormat, string> = { avif: "avif", webp: "webp", jpeg: "jpg", png: "png", gif: "gif", svg: "svg" };
export const MIME_BY_EXT: Record<string, string> = {
  avif: "image/avif", webp: "image/webp", jpg: "image/jpeg", png: "image/png", gif: "image/gif", svg: "image/svg+xml",
};
export interface PipelineContext {
  config: ImagesConfig;
  processor: ImageProcessor;
  base: string;
  assetsDir: string;
  dev: boolean;
  cacheDir: string;
}
export function createPipeline(ctx: PipelineContext) {
  const cache = diskCache(ctx.cacheDir);
  const run = pool();
  async function placeholder(input: Buffer, digest: string, kind: "blur" | "dominant-color"): Promise<Placeholder> {
    const name = `${hash(digest, kind, ctx.processor.id)}.placeholder.json`;
    const cached = await cache.get(name);
    if (cached) return JSON.parse(cached.toString("utf8")) as Placeholder;
    const value = await ctx.processor.placeholder(input, kind);
    await cache.set(name, Buffer.from(JSON.stringify(value)));
    return value;
  }
  async function describeFile(file: string, directives: Directives & { preset?: string }): Promise<Described> {
    const options = resolveOptions(ctx.config, directives, file);
    const input = await readFile(file);
    const digest = hash(input);
    const meta = await ctx.processor.metadata(input);
    const stem = basename(file, extname(file)).replace(/[^\w.-]/g, "_");
    const asset = (key: string, ext: string, width: number, encode?: EncodeOptions): PlannedAsset => {
      const fileName = `${ctx.assetsDir}/${stem}-${width}-${key.slice(0, 8)}.${ext}`;
      return { key, ext, file, fileName, url: ctx.dev ? `${ctx.base}@rivet/image/${key}.${ext}` : ctx.base + fileName, ...(encode ? { encode } : {}) };
    };
    if (meta.format === "svg" || meta.animated) {
      if (meta.format === "tiff") throw new Error(`${file}: animated TIFF is not supported`);
      const width = options.width ?? meta.width;
      const height = options.height ?? Math.round((width * meta.height) / meta.width);
      const a = asset(digest, EXT[meta.format], width);
      return { descriptor: single(a.url, meta.format, width, height, options.layout), assets: [a], warnings: [] };
    }
    const plan = planWidths({ intrinsic: meta, layout: options.layout, width: options.width, height: options.height, densities: options.densities, breakpoints: options.breakpoints });
    const warnings = plan.clamped ? [`${file}: requested width ${options.width}px exceeds intrinsic ${meta.width}px; clamped`] : [];
    const fallback = fallbackFormat(meta.format, meta.hasAlpha);
    const formats = [...new Set([...options.formats.filter((f) => f !== fallback), fallback])];
    const assets: PlannedAsset[] = [];
    const sources = formats.map((format) => ({
      format,
      urls: plan.widths.map(({ width }) => {
        const encode: EncodeOptions = {
          width,
          ...(options.height ? { height: Math.round((width * plan.height) / plan.width) } : {}),
          format,
          quality: options.quality[format],
          fit: options.fit,
          position: options.position,
        };
        const a = asset(hash(digest, JSON.stringify(encode), ctx.processor.id), EXT[format], width, encode);
        assets.push(a);
        return a.url;
      }),
    }));
    const ph = options.placeholder === "none" ? undefined : await placeholder(input, digest, options.placeholder);
    return { descriptor: describe({ plan, layout: options.layout, placeholder: ph, sources }), assets, warnings };
  }
  async function encode(asset: PlannedAsset): Promise<Buffer> {
    const name = `${asset.key}.${asset.ext}`;
    const cached = await cache.get(name);
    if (cached) return cached;
    return run(async () => {
      const input = await readFile(asset.file);
      const output = asset.encode ? await ctx.processor.encode(input, asset.encode) : input;
      await cache.set(name, output);
      return output;
    });
  }
  return { describe: describeFile, encode };
}
```

```ts
// libs/compiler/images/plugin.ts
import { resolve } from "node:path";
import type { Plugin, ResolvedConfig } from "vite";
import { parseImageId, validateConfig, type ImagesConfig } from "./options.ts";
import { sharpProcessor } from "./processor.ts";
import { createPipeline, MIME_BY_EXT, type PlannedAsset } from "./pipeline.ts";

const RUNTIME = "@rivetjs/runtime/image";
const SERVER_ONLY_WARNING =
  "image variants referenced from server code may be missing from the client build; build with createBuilder().buildApp() (vite build --app)";
/** `?image` / `?jsx` imports. Shared across environments: every environment registers variants, the client emits them. */
export function imagesPlugin(input: ImagesConfig | true): Plugin {
  const config = input === true ? {} : input;
  validateConfig(config);
  const processor = config.processor ?? sharpProcessor();
  const registry = new Map<string, PlannedAsset>(); // `${key}.${ext}` → asset
  let pipeline: ReturnType<typeof createPipeline>;
  let resolved: ResolvedConfig;
  let appBuild = false;
  const isClient = (ctx: { environment?: { config: { consumer: string } } }, ssr?: boolean) =>
    ctx.environment ? ctx.environment.config.consumer === "client" : !ssr;
  return {
    name: "vite-plugin-rivet-images",
    enforce: "pre",
    sharedDuringBuild: true,
    configResolved(config_) {
      resolved = config_;
      pipeline = createPipeline({
        config,
        processor,
        base: config_.base,
        assetsDir: config_.build.assetsDir,
        dev: config_.command === "serve",
        cacheDir: resolve(config_.root, config.cacheDir ?? "node_modules/.rivet/images"),
      });
    },
    // Server-only images register during the ssr build, so it must run before the client emits.
    buildApp: {
      order: "pre",
      async handler(builder) {
        appBuild = true;
        const { ssr, client } = builder.environments;
        if (ssr && !ssr.isBuilt) await builder.build(ssr);
        if (client && !client.isBuilt) await builder.build(client);
      },
    },
    async resolveId(id, importer, options) {
      if (id === RUNTIME && isClient(this, options?.ssr)) return this.resolve(RUNTIME + "/dom", importer, { skipSelf: true });
      return null;
    },
    async load(id) {
      let request;
      try {
        request = parseImageId(id);
        if (!request) return null;
        this.addWatchFile(request.file);
        const { descriptor, assets, warnings } = await pipeline.describe(request.file, request.directives);
        for (const warning of warnings) this.warn(warning);
        for (const asset of assets) registry.set(`${asset.key}.${asset.ext}`, asset);
        const json = JSON.stringify(descriptor);
        return request.kind === "image"
          ? `export default ${json};\n`
          : `import { Picture } from "${RUNTIME}";\nconst image = ${json};\nexport default (props) => Picture({ ...props, image });\n`;
      } catch (error) {
        this.error({ message: (error as Error).message, id: request?.file ?? id });
      }
    },
    async generateBundle() {
      if (resolved.command !== "build" || !registry.size) return;
      if (!isClient(this)) {
        if (!appBuild) this.warn(SERVER_ONLY_WARNING);
        return;
      }
      const emitted = new Set<string>();
      await Promise.all(
        [...registry.values()].map(async (asset) => {
          if (emitted.has(asset.fileName)) return;
          emitted.add(asset.fileName);
          this.emitFile({ type: "asset", fileName: asset.fileName, source: await pipeline.encode(asset) });
        }),
      );
    },
    configureServer(server) {
      const prefix = resolved.base + "@rivet/image/";
      server.middlewares.use(async (req, res, next) => {
        const path = req.url?.split("?")[0];
        if (!path?.startsWith(prefix)) return next();
        const asset = registry.get(path.slice(prefix.length));
        if (!asset) {
          res.statusCode = 404;
          return res.end();
        }
        const etag = `"${asset.key}"`;
        if (req.headers["if-none-match"] === etag) {
          res.statusCode = 304;
          return res.end();
        }
        try {
          const body = await pipeline.encode(asset);
          res.setHeader("Content-Type", MIME_BY_EXT[asset.ext]!);
          res.setHeader("Cache-Control", "no-cache");
          res.setHeader("ETag", etag);
          res.end(body);
        } catch (error) {
          next(error);
        }
      });
    },
  };
}
```

```ts
// libs/compiler/images/index.ts
import { mkdir, writeFile } from "node:fs/promises";
import { dirname, join, resolve } from "node:path";
import type { ImageDescriptor } from "@rivetjs/runtime/image-plan";
import { checkDirectives, validateConfig, type Directives, type ImagesConfig } from "./options.ts";
import { sharpProcessor } from "./processor.ts";
import { createPipeline } from "./pipeline.ts";
export { sharpProcessor, type ImageProcessor, type ImageMetadata, type EncodeOptions } from "./processor.ts";
export type { ImagesConfig, Directives } from "./options.ts";

export interface GetImageOptions {
  /** Directory served at `base` (e.g. the client build output). */
  outDir: string;
  base?: string;
  assetsDir?: string;
  config?: ImagesConfig;
  directives?: Directives & { preset?: string };
}
/** Process a file known only at runtime/prerender (CMS uploads); same descriptor as `?image`. */
export async function getImage(file: string, options: GetImageOptions): Promise<ImageDescriptor> {
  const config = options.config ?? {};
  validateConfig(config);
  const directives = checkDirectives(options.directives ?? {}, file);
  const pipeline = createPipeline({
    config,
    processor: config.processor ?? sharpProcessor(),
    base: options.base ?? "/",
    assetsDir: options.assetsDir ?? "assets",
    dev: false,
    cacheDir: resolve(config.cacheDir ?? "node_modules/.rivet/images"),
  });
  const { descriptor, assets } = await pipeline.describe(resolve(file), directives);
  await Promise.all(
    assets.map(async (asset) => {
      const target = join(options.outDir, asset.fileName);
      await mkdir(dirname(target), { recursive: true });
      await writeFile(target, await pipeline.encode(asset));
    }),
  );
  return descriptor;
}
```
