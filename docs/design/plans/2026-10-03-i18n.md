# i18n Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Compile-time i18n for Rivet: `t("key", params)` inlined into server fragment arrays (one per locale) and per-locale client chunks, with an isomorphic locale resolver and router integration.

**Architecture:** New `@rivetjs/i18n` parses ICU catalogs into a per-key, per-locale table at build time and registers it with the Rust/wasm compiler through a new declarative static-call hook. The compiler lowers matched `t()` calls: server target emits `_variants([...])` fragment sets and `_variantValue([...])` slot values selected by `RenderOptions.variant`; DOM target is compiled once per locale with literals inlined. Router picks locale (strategy array), passes `variant` + that locale's resume manifest to the renderer.

**Tech Stack:** TypeScript (Node 24 type stripping, `node --test`), Rust 2024 + oxc 0.75 → wasm32 (`libs/compiler/rust`, built by `rust/build.sh` in Docker), Vite 8 plugin API, `@formatjs/icu-messageformat-parser` 2.11 (build-time only), Playwright 1.63.

**Spec:** `docs/design/specs/2026-10-03-i18n-design.md`

## Global Constraints

- Node `>=24`; packages `0.0.0-experimental`, `private: true`, `type: module`, `sideEffects: false`, exports point at `.ts` sources (`types` + `default`).
- `tsconfig.base.json`: `erasableSyntaxOnly` — no enums, namespaces, parameter properties. Imports use explicit `.ts` extensions.
- Zero-cost rule: compiler output byte-identical when `staticCalls` absent or unmatched. ssr-bench Razor comparison must not move.
- No catalog, ICU parser or message runtime shipped to the browser for static messages; no resume-state entries for messages.
- Variant index never in module/global state on the server (concurrent requests + awaits) — only in `renderCore` closure.
- Keys: string literal or prefixed template literal only; anything else = compile diagnostic.
- Missing key in default locale = build error always. `missing` policy (`"error" | "warn" | "fallback"`) applies to non-default locales.
- Rust changes require rebuilding and committing `libs/compiler/rivet_compiler.wasm` (`npm run build:wasm -w @rivetjs/compiler`, needs Docker).
- Every task: `npm run typecheck` and the touched project's `test` target pass before commit. Format with `npx oxfmt` on touched TS files.
- Commit messages end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```ts
// ICU MessageFormat → compact IR used by the compiler hook. Build-time only.
import { parse, TYPE, type MessageFormatElement } from "@formatjs/icu-messageformat-parser";

export type Part =
  | { t: string }
  | { a: string }
  | { n: string; f?: Intl.NumberFormatOptions }
  | { d: string; f?: Intl.DateTimeFormatOptions }
  | { p: string; ord?: true; cases: Record<string, Part[]> }
  | { s: string; cases: Record<string, Part[]> }
  | { tag: string; c: Part[] };
export type Variant = string | Part[];
export type ParamKind = "string" | "number" | "date" | "select" | "tag";

export class I18nError extends Error {
  override name = "I18nError";
}

const NUMBER_STYLES: Record<string, Intl.NumberFormatOptions> = {
  percent: { style: "percent" },
  integer: { maximumFractionDigits: 0 },
};
const DATE_STYLES: Record<string, Intl.DateTimeFormatOptions> = {
  short: { dateStyle: "short" },
  medium: { dateStyle: "medium" },
  long: { dateStyle: "long" },
  full: { dateStyle: "full" },
};

function convert(elements: MessageFormatElement[], where: string, pound: string | null): Part[] {
  const out: Part[] = [];
  const text = (s: string) => {
    const last = out[out.length - 1];
    if (last && "t" in last) last.t += s;
    else out.push({ t: s });
  };
  for (const el of elements) {
    switch (el.type) {
      case TYPE.literal: text(el.value); break;
      case TYPE.argument: out.push({ a: el.value }); break;
      case TYPE.number: {
        const style = typeof el.style === "string" ? NUMBER_STYLES[el.style] : el.style ? (el.style as { parsedOptions: Intl.NumberFormatOptions }).parsedOptions : undefined;
        if (typeof el.style === "string" && !style) throw new I18nError(`${where}: unknown number style "${el.style}"`);
        out.push(style ? { n: el.value, f: style } : { n: el.value });
        break;
      }
      case TYPE.date:
      case TYPE.time: {
        const style = typeof el.style === "string" ? DATE_STYLES[el.style] : el.style ? (el.style as { parsedOptions: Intl.DateTimeFormatOptions }).parsedOptions : undefined;
        if (typeof el.style === "string" && !style) throw new I18nError(`${where}: unknown date style "${el.style}"`);
        const f = el.type === TYPE.time && style && "dateStyle" in style ? { timeStyle: style.dateStyle } : style;
        out.push(f ? { d: el.value, f: f as Intl.DateTimeFormatOptions } : { d: el.value });
        break;
      }
      case TYPE.pound:
        if (!pound) throw new I18nError(`${where}: "#" outside plural`);
        out.push({ n: pound });
        break;
      case TYPE.plural:
      case TYPE.select: {
        if (!el.options.other) throw new I18nError(`${where}: {${el.value}} requires an "other" case`);
        if (el.type === TYPE.plural && el.offset) throw new I18nError(`${where}: plural offset is not supported`);
        const cases: Record<string, Part[]> = {};
        for (const [k, option] of Object.entries(el.options)) cases[k] = convert(option.value, where, el.type === TYPE.plural ? el.value : pound);
        out.push(el.type === TYPE.select ? { s: el.value, cases } : el.pluralType === "ordinal" ? { p: el.value, ord: true, cases } : { p: el.value, cases });
        break;
      }
      case TYPE.tag: out.push({ tag: el.value, c: convert(el.children, where, pound) }); break;
    }
  }
  return out;
}

export function parseMessage(message: string, where: string): Variant {
  let ast: MessageFormatElement[];
  try {
    ast = parse(message, { shouldParseSkeletons: true, requiresOtherClause: false });
  } catch (error) {
    throw new I18nError(`${where}: ${(error as Error).message}`);
  }
  const parts = convert(ast, where, null);
  if (parts.length === 0) return "";
  if (parts.length === 1 && "t" in parts[0]!) return parts[0].t;
  return parts;
}

export function paramsOf(v: Variant, into = new Map<string, ParamKind>()): Map<string, ParamKind> {
  if (typeof v === "string") return into;
  for (const part of v) {
    if ("a" in part) into.set(part.a, into.get(part.a) ?? "string");
    else if ("n" in part) into.set(part.n, "number");
    else if ("d" in part) into.set(part.d, "date");
    else if ("p" in part) { into.set(part.p, "number"); for (const c of Object.values(part.cases)) paramsOf(c, into); }
    else if ("s" in part) { into.set(part.s, "select"); for (const c of Object.values(part.cases)) paramsOf(c, into); }
    else if ("tag" in part) { into.set(part.tag, "tag"); paramsOf(part.c, into); }
  }
  return into;
}
```
Note: `{ n: "n" }` for `#` overrides an earlier `"string"` guess to `"number"` because `set` overwrites — intended.

```ts
// Catalog loading and the per-key, per-locale table handed to the compiler.
import { readFile } from "node:fs/promises";
import { resolve } from "node:path";
import { I18nError, paramsOf, parseMessage, type Variant } from "./icu.ts";

export interface I18nConfig {
  locales: readonly string[];
  defaultLocale: string;
  catalogs: string | ((locale: string) => Record<string, unknown> | Promise<Record<string, unknown>>);
  fallback?: Readonly<Record<string, string>>;
  missing?: "error" | "warn" | "fallback";
  routing?: { prefix: "always" | "as-needed" | "never" | Readonly<Record<string, string>> };
  strategy?: readonly string[];
  cookie?: { name?: string; maxAge?: number } | false;
  timeZone?: string;
  baseUrl?: string;
}
export type Table = Record<string, Variant[]>;
export interface BuiltTable {
  table: Table;
  warnings: string[];
  files: string[];
}

export function flatten(tree: Record<string, unknown>, prefix = ""): Record<string, string> {
  const out: Record<string, string> = {};
  for (const [k, v] of Object.entries(tree)) {
    const key = prefix ? prefix + "." + k : k;
    if (typeof v === "string") out[key] = v;
    else if (v && typeof v === "object" && !Array.isArray(v)) Object.assign(out, flatten(v as Record<string, unknown>, key));
    else throw new I18nError(`${key}: catalog values must be strings or nested objects`);
  }
  return out;
}

export function fallbackChain(locale: string, config: I18nConfig): string[] {
  const chain: string[] = [];
  for (let l: string | undefined = locale; l && !chain.includes(l); l = config.fallback?.[l]) chain.push(l);
  if (!chain.includes(config.defaultLocale)) chain.push(config.defaultLocale);
  return chain;
}

export function buildTable(config: I18nConfig, raw: Record<string, Record<string, unknown>>): BuiltTable {
  const { locales, defaultLocale } = config;
  if (!locales.includes(defaultLocale)) throw new I18nError(`defaultLocale "${defaultLocale}" is not in locales`);
  const missing = config.missing ?? "error";
  const flat = Object.fromEntries(locales.map((l) => [l, flatten(raw[l] ?? {})]));
  const keys = new Set(locales.flatMap((l) => Object.keys(flat[l]!)));
  const table: Table = {};
  const warnings: string[] = [];
  for (const key of [...keys].sort()) {
    if (flat[defaultLocale]![key] === undefined) throw new I18nError(`${key}: missing in default locale "${defaultLocale}"`);
    const parsed = new Map<string, Variant>();
    table[key] = locales.map((locale) => {
      const chain = fallbackChain(locale, config);
      const found = chain.find((l) => flat[l]![key] !== undefined)!;
      if (found !== locale) {
        if (missing === "error") throw new I18nError(`${key}: missing in "${locale}"`);
        if (missing === "warn") warnings.push(`${key}: missing in "${locale}", using "${found}"`);
      }
      let v = parsed.get(found);
      if (v === undefined) parsed.set(found, (v = parseMessage(flat[found]![key]!, `${found}:${key}`)));
      return v;
    });
    const base = [...paramsOf(table[key][0]!).keys()].sort().join(",");
    locales.forEach((locale, i) => {
      const own = [...paramsOf(table[key]![i]!).keys()].sort().join(",");
      if (own !== base) throw new I18nError(`${key}: params differ between "${defaultLocale}" (${base}) and "${locale}" (${own})`);
    });
  }
  return { table, warnings, files: [] };
}

export async function loadCatalogs(config: I18nConfig, root: string): Promise<{ raw: Record<string, Record<string, unknown>>; files: string[] }> {
  const raw: Record<string, Record<string, unknown>> = {};
  const files: string[] = [];
  for (const locale of config.locales) {
    if (typeof config.catalogs === "function") raw[locale] = await config.catalogs(locale);
    else {
      const file = resolve(root, config.catalogs.replaceAll("{locale}", locale));
      files.push(file);
      raw[locale] = JSON.parse(await readFile(file, "utf8")) as Record<string, unknown>;
    }
  }
  return { raw, files };
}
```

import { paramsOf, type Part, type Variant } from "./icu.ts";
import type { Table } from "./catalog.ts";

function selectCases(v: Variant, name: string, into: Set<string>): void {
  if (typeof v === "string") return;
  for (const part of v as Part[]) {
    if ("s" in part && part.s === name) for (const k of Object.keys(part.cases)) if (k !== "other") into.add(k);
    for (const nested of "cases" in part ? Object.values(part.cases) : "c" in part ? [part.c] : []) selectCases(nested, name, into);
  }
}

export function declarations(table: Table, locales: readonly string[]): string {
  const lines = ["// Generated by @rivetjs/i18n. Do not edit.", 'declare module "@rivetjs/i18n" {', "  interface Messages {"];
  for (const [key, variants] of Object.entries(table)) {
    const params = paramsOf(variants[0]!);
    const fields = [...params].sort(([a], [b]) => a.localeCompare(b)).map(([name, kind]) => {
      let type: string;
      if (kind === "number") type = "number";
      else if (kind === "date") type = "Date | number";
      else if (kind === "tag") type = "(children: unknown) => unknown";
      else if (kind === "select") {
        const cases = new Set<string>();
        for (const v of variants) selectCases(v, name, cases);
        type = [...[...cases].sort().map((c) => JSON.stringify(c)), "(string & {})"].join(" | ");
      } else type = "string | number";
      return `${name}: ${type}`;
    });
    lines.push(`    ${JSON.stringify(key)}: ${fields.length ? `{ ${fields.join("; ")} }` : "{}"};`);
  }
  lines.push("  }", "  interface Locales {", ...locales.map((l) => `    ${JSON.stringify(l)}: true;`), "  }", "}", "export {};", "");
  return lines.join("\n") + `// export type Locale = ${locales.map((l) => JSON.stringify(l)).join(" | ")}\n`;
}

```ts
// Message runtime used by compiled `t()` code: only what depends on runtime values.
// Intl constructors are ~50x the cost of format(); every formatter is cached per locale + options.
const pluralRules = new Map<string, Intl.PluralRules>();
const formatters = new Map<string, Intl.NumberFormat | Intl.DateTimeFormat>();

function rules(locale: string, ordinal: boolean): Intl.PluralRules {
  const key = (ordinal ? "o:" : "c:") + locale;
  let r = pluralRules.get(key);
  if (!r) pluralRules.set(key, (r = new Intl.PluralRules(locale, { type: ordinal ? "ordinal" : "cardinal" })));
  return r;
}

export function plural(locale: string, n: number, cases: Readonly<Record<string, () => string>>, ordinal = false): string {
  const exact = cases["=" + n];
  if (exact) return exact();
  return (cases[rules(locale, ordinal).select(n)] ?? cases.other!)();
}

export function select(value: unknown, cases: Readonly<Record<string, () => string>>): string {
  const key = String(value);
  return (Object.hasOwn(cases, key) ? cases[key]! : cases.other!)();
}

export function number(locale: string, n: number, options?: Intl.NumberFormatOptions): string {
  const key = "n:" + locale + ":" + (options ? JSON.stringify(options) : "");
  let f = formatters.get(key) as Intl.NumberFormat | undefined;
  if (!f) formatters.set(key, (f = new Intl.NumberFormat(locale, options)));
  return f.format(n);
}

export function date(locale: string, d: Date | number, options?: Intl.DateTimeFormatOptions, timeZone?: string): string {
  const key = "d:" + locale + ":" + (timeZone ?? "") + ":" + (options ? JSON.stringify(options) : "");
  let f = formatters.get(key) as Intl.DateTimeFormat | undefined;
  if (!f) formatters.set(key, (f = new Intl.DateTimeFormat(locale, timeZone ? { ...options, timeZone } : options)));
  return f.format(d);
}

export function pick<T>(locale: string, locales: readonly string[], values: readonly (() => T)[]): T {
  const i = locales.indexOf(locale);
  return values[i < 0 ? 0 : i]!();
}
```

struct ServerTemplate {
    /// One fragment list + slot order per variant (one entry when no message differs per variant).
    sets: Vec<(Vec<String>, Vec<usize>)>,
    slots: Vec<Expr>,
    model_temps: Vec<String>,
}
impl ServerTemplate {
    fn new(variants: usize) -> Self {
        Self { sets: (0..variants.max(1)).map(|_| (vec![String::new()], Vec::new())).collect(), slots: Vec::new(), model_temps: Vec::new() }
    }
    fn append(&mut self, text: &str) { for (f, _) in &mut self.sets { f.last_mut().unwrap().push_str(text); } }
    fn append_variant(&mut self, v: usize, text: &str) { self.sets[v].0.last_mut().unwrap().push_str(text); }
    fn slot(&mut self, value: Expr) { let i = self.add_slot(value); for v in 0..self.sets.len() { self.emit_slot(v, i); } }
    fn add_slot(&mut self, value: Expr) -> usize { self.slots.push(value); self.slots.len() - 1 }
    fn emit_slot(&mut self, v: usize, i: usize) { let (f, o) = &mut self.sets[v]; o.push(i); f.push(String::new()); }
    fn uniform(&self) -> bool { self.sets.windows(2).all(|w| w[0] == w[1]) }
    fn identity_orders(&self) -> bool { self.sets.iter().all(|(_, o)| o.iter().enumerate().all(|(k, &i)| k == i) && o.len() == self.slots.len()) }
}

            Part::Date { d, f } => format!("{}({}, {}, {}, {})", names.date, crate::ir::quote(locale), param(d), f.as_ref().map(|f| f.to_string()).unwrap_or("undefined".into()), tz.map(crate::ir::quote).unwrap_or("undefined".into())),
            Part::Plural { p, ord, cases } => format!("{}({}, {}, {{ {} }}{})", names.plural, crate::ir::quote(locale), param(p), case_object(cases, locale, param, names, tz)?, if ord.unwrap_or(false) { ", true" } else { "" }),
            Part::Select { s, cases } => format!("{}({}, {{ {} }})", names.select, param(s), case_object(cases, locale, param, names, tz)?),
            Part::Tag { .. } => return Err("Tags inside plural/select cases are not supported".into()),
        });
    }
    Ok(if out.is_empty() { "\"\"".into() } else { format!("\"\" + {}", out.join(" + ")) })
}
fn case_object(cases: &BTreeMap<String, Vec<Part>>, locale: &str, param: &dyn Fn(&str) -> String, names: &Names, tz: Option<&str>) -> Result<String, String> {
    cases.iter().map(|(k, v)| Ok(format!("{}: () => {}", crate::ir::quote(k), string_expr(v, locale, param, names, tz)?))).collect::<Result<Vec<_>, String>>().map(|v| v.join(", "))
}
pub struct Names { pub plural: String, pub select: String, pub number: String, pub date: String }
```
`Names` locals come from `self.i18n_runtime_name("plural")` etc., each recorded for the `@rivetjs/i18n/runtime` import line.

Param binding: before lowering a message, hoist each param expression into a per-template temp so it is evaluated once even when referenced by several variants' thunks:
- server: params become `t.add_slot(expr)` for `{a}`-only params (written directly); for params consumed by formatter thunks, declare a temp in `t.model_temps` (existing mechanism printing `let __rivetM1, …;` and wrapping the template in an IIFE — `lower()` `:1059-1066`), assign it in source order via an `Expr::Assign` slot that writes nothing (`(temp = expr, undefined)` slot — renderer skips `undefined`), then reference `temp` in thunks.
- DOM: params referenced inside `bindChild(dest, () => …)` thunks use the original expression text (reactivity requires re-evaluation inside the thunk) — the thunk *is* the single evaluation site per update.

Top-level part sequence per variant (server, text context): iterate `Variant::Parts` parts; `Text` → `append_variant(v, escaped)`; `Arg` → `emit_slot(v, slot_of(name))`; others → one slot holding `__rivetVariantValue([() => expr_en, () => expr_de])` created once per (message call, top-level part index) — requires all variants to have a formatter part at a corresponding position; when the variants differ structurally, fall back to **whole-message** variant value: single slot `__rivetVariantValue([() => string_expr(en), () => string_expr(de)])`. Attribute context: always the whole-message variant value as an `Expr::Attr` value (runtime resolves, Task 5).

DOM per variant: text parts → `t.html`; every other part → range slot + `bindChild(dest, () => <string_expr of that part>)`.

`t.in` → `Expr::Raw(format!("{}({}, [{}], [{}])", names.pick, locale_src, quoted_variants, thunks))`.

Param validation: compare `paramsOf`-equivalent set computed in Rust (`fn params(v: &Variant) -> BTreeSet<String>`) against object-literal keys.

// i18n Vite plugin: catalogs → compiler table, generated declarations, one client build per locale.
import { writeFile, mkdir } from "node:fs/promises";
import { dirname, resolve } from "node:path";
import type { Plugin } from "vite";
import { registerTable, type CompileOptions } from "@rivetjs/compiler";
import { buildTable, loadCatalogs, type I18nConfig } from "./catalog.ts";
import { declarations } from "./types.ts";

export interface I18nPluginApi {
  rivetCompileOptions(server: boolean): Partial<CompileOptions>;
  readonly config: I18nConfig;
  readonly locale: string | null;
}
const TABLE = "rivet-i18n";

export function i18n(options: I18nConfig & { dts?: string }): Plugin & { api: I18nPluginApi } {
  const { dts = "src/rivet-i18n.d.ts", ...config } = options;
  let root = process.cwd(), files: string[] = [], serve = false;
  const locale = process.env.RIVET_LOCALE ?? null;
  const users = new Set<string>();
  const refresh = async () => {
    const loaded = await loadCatalogs(config, root);
    files = loaded.files;
    const { table, warnings } = buildTable(serve && !config.missing ? { ...config, missing: "warn" } : config, loaded.raw);
    for (const w of warnings) console.warn("[rivet i18n] " + w);
    registerTable(TABLE, table);
    const file = resolve(root, dts);
    await mkdir(dirname(file), { recursive: true });
    await writeFile(file, declarations(table, config.locales));
  };
  const api: I18nPluginApi = {
    config,
    locale,
    rivetCompileOptions(server) {
      const base = { staticCalls: [{ id: TABLE, module: "@rivetjs/i18n", export: "t", prefixLookup: true }], variants: [...config.locales], ...(config.timeZone ? { timeZone: config.timeZone } : {}) };
      return server ? base : { ...base, variant: Math.max(0, config.locales.indexOf(locale ?? config.defaultLocale)) };
    },
  };
  return {
    name: "rivet-i18n",
    enforce: "pre",
    api,
    config(user, env) {
      serve = env.command === "serve";
      if (serve || user.build?.ssr) return;
      if (!locale) throw new Error("i18n: client builds need RIVET_LOCALE (one build per locale)");
      if (!config.locales.includes(locale)) throw new Error(`i18n: RIVET_LOCALE "${locale}" is not in locales`);
      const outDir = user.build?.outDir ?? "dist/client";
      return { base: (user.base ?? "/") + locale + "/", build: { outDir: resolve(outDir, locale) } };
    },
    async configResolved(resolved) {
      root = resolved.root;
      if (new Intl.DateTimeFormat("es", { month: "long" }).format(new Date(2026, 0, 1)) !== "enero")
        console.warn("[rivet i18n] Node lacks full ICU: dates and numbers will format in English");
      await refresh();
    },
    transform(code, id) {
      if (code.includes("@rivetjs/i18n")) users.add(id);
      return null;
    },
    async handleHotUpdate({ file, server }) {
      if (!files.includes(file)) return;
      await refresh();
      for (const id of users) {
        const mod = server.moduleGraph.getModuleById(id);
        if (mod) server.moduleGraph.invalidateModule(mod);
      }
      server.ws.send({ type: "full-reload" });
      return [];
    },
  };
}

export interface Source { path(): string; cookie(name: string): string | undefined; languages(): readonly string[] }
export interface Strategy { name: string; server?(req: Request, source: Source): string | undefined | Promise<string | undefined>; client?(source: Source): string | undefined }
export interface ResolveConfig { locales: readonly string[]; defaultLocale: string; routing?: I18nConfig["routing"]; cookie?: I18nConfig["cookie"] }

const DEFAULT_ORDER = ["url", "cookie", "preferredLanguage", "baseLocale"] as const;
const cookieName = (c: ResolveConfig) => (c.cookie === false ? null : (c.cookie?.name ?? "rivet_locale"));

export function match(wanted: readonly string[], locales: readonly string[]): string | undefined {
  const lower = locales.map((l) => l.toLowerCase());
  for (const w of wanted) {
    const want = w.toLowerCase();
    const exact = lower.indexOf(want);
    if (exact >= 0) return locales[exact];
    const base = lower.indexOf(want.split("-")[0]!);
    if (base >= 0) return locales[base];
    const region = lower.findIndex((l) => l.split("-")[0] === want.split("-")[0]);
    if (region >= 0) return locales[region];
  }
  return undefined;
}

function prefixes(config: ResolveConfig): [string, string][] {
  const p = config.routing?.prefix;
  if (p && typeof p === "object") return Object.entries(p).map(([l, path]) => [l, path.replace(/^\/|\/$/g, "")]);
  return config.locales.map((l) => [l, l.toLowerCase()]);
}
export function localeFromPath(path: string, config: ResolveConfig): string | undefined {
  if (config.routing?.prefix === "never") return undefined;
  const first = path.split("/")[1]?.toLowerCase() ?? "";
  return prefixes(config).find(([, p]) => p === first)?.[0];
}

const parseCookie = (header: string, name: string) => header.split(/;\s*/).map((p) => p.split("=")).find(([k]) => k === name)?.[1];
export function requestSource(req: Request): Source {
  const url = new URL(req.url);
  return {
    path: () => url.pathname,
    cookie: (name) => { const v = parseCookie(req.headers.get("cookie") ?? "", name); return v === undefined ? undefined : decodeURIComponent(v); },
    languages: () =>
      (req.headers.get("accept-language") ?? "")
        .split(",")
        .map((part) => { const [tag, ...params] = part.trim().split(";"); const q = params.find((p) => p.trim().startsWith("q=")); return { tag: tag!.trim(), q: q ? Number(q.trim().slice(2)) : 1 }; })
        .filter((l) => l.tag && l.tag !== "*" && l.q > 0)
        .sort((a, b) => b.q - a.q)
        .map((l) => l.tag),
  };
}
export function browserSource(): Source {
  return {
    path: () => location.pathname,
    cookie: (name) => { const v = parseCookie(document.cookie, name); return v === undefined ? undefined : decodeURIComponent(v); },
    languages: () => navigator.languages ?? [navigator.language],
  };
}

function builtins(config: ResolveConfig): Record<(typeof DEFAULT_ORDER)[number], Strategy> {
  const fromPath = (s: Source) => localeFromPath(s.path(), config);
  const fromCookie = (s: Source) => { const n = cookieName(config); return n ? s.cookie(n) : undefined; };
  const fromLanguages = (s: Source) => match(s.languages(), config.locales);
  return {
    url: { name: "url", server: (_r, s) => fromPath(s), client: fromPath },
    cookie: { name: "cookie", server: (_r, s) => fromCookie(s), client: fromCookie },
    preferredLanguage: { name: "preferredLanguage", server: (_r, s) => fromLanguages(s), client: fromLanguages },
    baseLocale: { name: "baseLocale", server: () => config.defaultLocale, client: () => config.defaultLocale },
  };
}
const strategyList = (config: ResolveConfig, order: readonly (string | Strategy)[]) => {
  const b = builtins(config);
  return order.map((s) => { if (typeof s !== "string") return s; const found = b[s as keyof typeof b]; if (!found) throw new Error(`i18n: unknown strategy "${s}"`); return found; });
};

export async function resolveServer(req: Request, config: ResolveConfig, order: readonly (string | Strategy)[] = DEFAULT_ORDER): Promise<string> {
  const source = requestSource(req);
  for (const s of strategyList(config, order)) {
    const l = await s.server?.(req, source);
    if (l && config.locales.includes(l)) return l;
  }
  return config.defaultLocale;
}
export function resolveClient(config: ResolveConfig, order: readonly (string | Strategy)[] = DEFAULT_ORDER): string {
  const source = browserSource();
  for (const s of strategyList(config, order)) {
    const l = s.client?.(source);
    if (l && config.locales.includes(l)) return l;
  }
  return config.defaultLocale;
}
export const strategies = builtins;
```

// @rivetjs/router integration: locale guard, href helpers, SEO alternates.
import { notFound, redirect, type RequestEvent } from "@rivetjs/router";
import type { HeadLink } from "@rivetjs/router";   // check the actual export path of HeadLink in libs/router/index.ts
import { localeFromPath, resolveServer, type ResolveConfig, type Strategy } from "./resolve.ts";

export interface I18nRouteConfig extends ResolveConfig {
  baseUrl?: string;
  strategy?: readonly (string | Strategy)[];
  rtl?: readonly string[];
}
const RTL = ["ar", "fa", "he", "ur", "ps", "dv", "yi"];
export const dirOf = (locale: string, rtl: readonly string[] = RTL): "ltr" | "rtl" => (rtl.includes(locale.split("-")[0]!.toLowerCase()) ? "rtl" : "ltr");

function prefixOf(locale: string, config: I18nRouteConfig): string {
  const p = config.routing?.prefix ?? "as-needed";
  if (p === "never") return "";
  if (typeof p === "object") return p[locale] ?? "/" + locale;
  if (p === "as-needed" && locale === config.defaultLocale) return "";
  return "/" + locale.toLowerCase();
}
export function delocalizeHref(href: string, config: I18nRouteConfig): string {
  const url = new URL(href, "http://x");
  const found = localeFromPath(url.pathname, config);
  if (found) url.pathname = url.pathname.replace(/^\/[^/]+/, "") || "/";
  return url.pathname + url.search + url.hash;
}
export function localizeHref(href: string, locale: string, config: I18nRouteConfig): string {
  const bare = delocalizeHref(href, config);
  const prefix = prefixOf(locale, config);
  return prefix + (bare === "/" && prefix ? "" : bare) || "/";
}
export function alternates(url: URL, config: I18nRouteConfig): HeadLink[] {
  if (!config.baseUrl) throw new Error("i18n: alternates() needs baseUrl");
  const bare = delocalizeHref(url.pathname, config);
  const abs = (l: string) => new URL(localizeHref(bare, l, config), config.baseUrl).href;
  return [
    ...config.locales.map((l) => ({ rel: "alternate", hreflang: l, href: abs(l) })),
    { rel: "alternate", hreflang: "x-default", href: abs(config.defaultLocale) },
    { rel: "canonical", href: new URL(url.pathname + url.search, config.baseUrl).href },
  ];
}
export function i18nGuard(config: I18nRouteConfig) {
  return async (event: RequestEvent): Promise<Response | void> => {
    const param = event.params.lang;
    if (param !== undefined && !config.locales.includes(param)) throw notFound();
    const locale = param ?? (await resolveServer(event.request, config, config.strategy));
    const prefix = config.routing?.prefix ?? "as-needed";
    if (event.url.pathname === "/" && prefix !== "never" && prefixOf(locale, config)) throw redirect(localizeHref("/", locale, config), 302);
    if (param === undefined && prefix !== "never" && locale !== config.defaultLocale && !localeFromPath(event.url.pathname, config)) {
      // Deep link without prefix under as-needed: serve default locale, don't bounce (Nuxt redirectOn: 'root').
      event.locals.locale = config.defaultLocale;
    } else event.locals.locale = locale;
    event.locals.dir = dirOf(event.locals.locale as string, config.rtl);
    event.setHeader("content-language", event.locals.locale as string);
    if (param === undefined && prefix === "never") event.setHeader("vary", "cookie, accept-language");
  };
}
